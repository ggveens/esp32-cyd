// TFT helpers — vẽ giao diện OTA trên màn hình
// ─────────────────────────────────────────────────────────────
#define OTA_BG       0x0841   // Giống BG_COLOR của ui.cpp
#define OTA_ACCENT   0x2D7F   // Cyan
#define OTA_SUCCESS  0x07E0   // Xanh lá
#define OTA_ERROR    0xF800   // Đỏ
#define OTA_BAR_BG   0x2104   // Thanh nền
#define OTA_BAR_FG   0x2D7F   // Thanh progress

static void _drawOTAScreen(const char* title, const char* subtitle, int percent) {
    // Chỉ vẽ khi cần — tránh SPI flood
    tft.fillScreen(OTA_BG);

    // Icon update (hình mũi tên đơn giản)
    tft.fillCircle(160, 60, 28, OTA_ACCENT);
    tft.fillCircle(160, 60, 22, OTA_BG);
    tft.fillTriangle(148, 68, 172, 68, 160, 48, OTA_ACCENT);
    tft.fillRect(155, 68, 10, 16, OTA_ACCENT);

    // Title
    tft.setTextSize(2);
    tft.setTextColor(0xFFFF, OTA_BG);
    int16_t tw = tft.textWidth(title);
    tft.setCursor((320 - tw) / 2, 100);
    tft.print(title);

    // Subtitle
    tft.setTextSize(1);
    tft.setTextColor(0x7BEF, OTA_BG);
    int16_t sw = tft.textWidth(subtitle);
    tft.setCursor((320 - sw) / 2, 124);
    tft.print(subtitle);

    // Progress bar background
    tft.fillRoundRect(20, 148, 280, 18, 4, OTA_BAR_BG);

    if (percent >= 0) {
        // Progress bar fill
        int barW = (int)(280.0f * percent / 100.0f);
        if (barW > 0) {
            tft.fillRoundRect(20, 148, barW, 18, 4, OTA_BAR_FG);
        }

        // Phần trăm
        tft.setTextSize(2);
        tft.setTextColor(0xFD20, OTA_BG);  // Màu cam
        char pct[8];
        snprintf(pct, sizeof(pct), "%d%%", percent);
        int16_t px = (320 - tft.textWidth(pct)) / 2;
        tft.fillRect(0, 172, 320, 20, OTA_BG);
        tft.setCursor(px, 174);
        tft.print(pct);
    }

    // Footer hint
    tft.setTextSize(1);
    tft.setTextColor(0x528A, OTA_BG);
    tft.setCursor(10, 210);
    tft.print("Khong ngat nguon trong khi cap nhat!");
}

static void _drawOTAResult(bool success, const char* msg) {
    tft.fillScreen(OTA_BG);

    uint16_t iconColor = success ? OTA_SUCCESS : OTA_ERROR;

    // Vòng tròn kết quả
    tft.fillCircle(160, 80, 35, iconColor);
    tft.fillCircle(160, 80, 28, OTA_BG);

    if (success) {
        // Dấu tick
        tft.drawLine(145, 80, 155, 92, iconColor);
        tft.drawLine(145, 81, 155, 93, iconColor);
        tft.drawLine(155, 92, 178, 66, iconColor);
        tft.drawLine(155, 93, 178, 67, iconColor);
    } else {
        // Dấu X
        tft.drawLine(148, 68, 172, 92, iconColor);
        tft.drawLine(149, 68, 173, 92, iconColor);
        tft.drawLine(172, 68, 148, 92, iconColor);
        tft.drawLine(173, 68, 149, 92, iconColor);
    }

    // Status text
    tft.setTextSize(2);
    tft.setTextColor(iconColor, OTA_BG);
    const char* statusText = success ? "THANH CONG!" : "THAT BAI";
    int16_t tw = tft.textWidth(statusText);
    tft.setCursor((320 - tw) / 2, 128);
    tft.print(statusText);

    // Message
    tft.setTextSize(1);
    tft.setTextColor(0x7BEF, OTA_BG);
    int16_t mw = tft.textWidth(msg);
    tft.setCursor((320 - mw) / 2, 155);
    tft.print(msg);

    if (success) {
        tft.setTextColor(0xFD20, OTA_BG);
        tft.setCursor(60, 178);
        tft.print("Dang khoi dong lai...");
    }
}

// Handler: GET /ota — trang upload
// ─────────────────────────────────────────────────────────────
static void handleOTAPage() {
    server.sendHeader("Cache-Control", "no-store");
    server.send_P(200, "text/html", OTA_HTML);
}

// ─────────────────────────────────────────────────────────────
// Handler: GET /ota/info — thông tin thiết bị cho trang OTA
// ─────────────────────────────────────────────────────────────
static void handleOTAInfo() {
    String mode = "offline";
    // Kiểm tra WiFi STA để xác định mode
    if (WiFi.status() == WL_CONNECTED) mode = "online";

    String json = "{";
    json += "\"deviceId\":\"" + deviceId + "\",";
    json += "\"mode\":\"" + mode + "\",";
    json += "\"freeHeap\":" + String(ESP.getFreeHeap()) + ",";
    json += "\"flashSize\":" + String(ESP.getFlashChipSize()) + ",";
    json += "\"sketchSize\":" + String(ESP.getSketchSize()) + ",";
    json += "\"sketchFree\":" + String(ESP.getFreeSketchSpace());
    json += "}";

    server.sendHeader("Cache-Control", "no-cache");
    server.send(200, "application/json", json);
}

// ─────────────────────────────────────────────────────────────
// Handler: POST /ota/upload — nhận và flash firmware
//
// ĐÂY LÀ HANDLER QUAN TRỌNG NHẤT — Được gọi 2 lần bởi WebServer:
//   1. Khi data chunk đến → ghi vào flash
//   2. Khi upload hoàn tất → gửi response
// ─────────────────────────────────────────────────────────────
static void handleOTAUpload() {
    // Lần gọi cuối — sau khi upload xong, gửi response
    if (Update.hasError()) {
        String errMsg = Update.errorString();
        Serial.println("[OTA] Thất bại: " + errMsg);
        _drawOTAResult(false, errMsg.c_str());
        server.send(500, "text/plain", "OTA FAILED: " + errMsg);
    } else {
        Serial.println(F("[OTA] Flash thành công — chuẩn bị restart"));
        _drawOTAResult(true, "Phien ban moi da duoc cai dat");
        server.send(200, "text/plain", "OK");
        server.client().flush();
        delay(1500);
        ESP.restart();
    }
    _otaInProgress = false;
}

static void handleOTAUploadData() {
    // Gọi mỗi khi có chunk data đến trong quá trình upload
    HTTPUpload& upload = server.upload();

    if (upload.status == UPLOAD_FILE_START) {
        Serial.printf("[OTA] Bắt đầu: %s (size chưa rõ)\n", upload.filename.c_str());
        _otaInProgress = true;

        // Tính space trống cho firmware
        uint32_t maxSketchSpace = (ESP.getFreeSketchSpace() - 0x1000) & 0xFFFFF000;

        if (!Update.begin(maxSketchSpace, U_FLASH)) {
            Serial.println("[OTA] Lỗi begin: " + String(Update.errorString()));
            _drawOTAScreen("OTA FAILED", Update.errorString(), -1);
            return;
        }

        _drawOTAScreen("Cap nhat firmware", "Dang ghi vao flash...", 0);
        Serial.println(F("[OTA] Update.begin() OK — đang nhận dữ liệu..."));

    } else if (upload.status == UPLOAD_FILE_WRITE) {
        // Ghi chunk vào flash
        if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
            Serial.println("[OTA] Lỗi write: " + String(Update.errorString()));
            return;
        }

        // Tính progress dùng Update.progress() / Update.size()
        // ESP32 HTTPUpload KHÔNG có remainingSize (field đó chỉ có trên ESP8266)
        // Update.progress() = bytes đã ghi vào flash (chính xác nhất)
        // Update.size()     = tổng size được khai báo lúc Update.begin()
        {
            uint32_t written = Update.progress();
            uint32_t total   = Update.size();
            int pct = (total > 0) ? (int)(100.0f * written / total) : 0;

            // Chỉ vẽ khi % thay đổi (bước 2%) để tránh SPI flood
            static int _lastPct = -1;
            if (pct != _lastPct && pct % 2 == 0) {
                _lastPct = pct;
                _drawOTAScreen("Cap nhat firmware", "Dang ghi vao flash...", pct);
            }
        }

    } else if (upload.status == UPLOAD_FILE_END) {
        if (Update.end(true)) {
            Serial.printf("[OTA] Hoàn tất: %u bytes → OK\n", upload.totalSize);
            _drawOTAScreen("Cap nhat firmware", "Flash thanh cong!", 100);
        } else {
            Serial.println("[OTA] Lỗi end: " + String(Update.errorString()));
        }

    } else if (upload.status == UPLOAD_FILE_ABORTED) {
        Update.abort();
        _otaInProgress = false;
        Serial.println(F("[OTA] Upload bị hủy"));
        _drawOTAResult(false, "Upload bi huy");
    }
}

// ─────────────────────────────────────────────────────────────
// ArduinoOTA setup — chỉ gọi 1 lần khi WiFi STA connected
// ─────────────────────────────────────────────────────────────
static void _setupArduinoOTA() {
    if (_arduinoOTASetup) return;

    // Hostname: <deviceId>.local → flash bằng lệnh:
    // pio run -t upload --upload-port <deviceId>.local
    ArduinoOTA.setHostname(deviceId.c_str());
    ArduinoOTA.setPassword(OTA_PASSWORD);

    ArduinoOTA.onStart([]() {
        String type = ArduinoOTA.getCommand() == U_FLASH ? "firmware" : "filesystem";
        Serial.println("[ArduinoOTA] Bắt đầu cập nhật: " + type);
        _otaInProgress = true;
        _drawOTAScreen("ArduinoOTA", ("Cap nhat " + type).c_str(), 0);
    });

    ArduinoOTA.onEnd([]() {
        Serial.println(F("\n[ArduinoOTA] Hoàn tất → Restart"));
        _drawOTAResult(true, "Khoi dong lai...");
        _otaInProgress = false;
    });

    ArduinoOTA.onProgress([](unsigned int progress, unsigned int total) {
        int pct = (int)(100.0f * progress / total);
        static int _lastPct = -1;
        if (pct != _lastPct && pct % 5 == 0) {
            _lastPct = pct;
            Serial.printf("[ArduinoOTA] %d%%\n", pct);
            _drawOTAScreen("ArduinoOTA", "Dang nhan du lieu...", pct);
        }
    });

    ArduinoOTA.onError([](ota_error_t error) {
        const char* errMsg;
        switch (error) {
            case OTA_AUTH_ERROR:    errMsg = "Sai mat khau"; break;
            case OTA_BEGIN_ERROR:   errMsg = "Loi Begin";    break;
            case OTA_CONNECT_ERROR: errMsg = "Loi Connect";  break;
            case OTA_RECEIVE_ERROR: errMsg = "Loi Receive";  break;
            case OTA_END_ERROR:     errMsg = "Loi End";      break;
            default:                errMsg = "Loi khong xac dinh";
        }
        Serial.printf("[ArduinoOTA] Lỗi [%u]: %s\n", error, errMsg);
        _drawOTAResult(false, errMsg);
        _otaInProgress = false;
    });

    ArduinoOTA.begin();
    _arduinoOTASetup = true;
    Serial.printf("[OTA] ArduinoOTA sẵn sàng: %s.local | pass: %s\n",
                  deviceId.c_str(), OTA_PASSWORD);
}

// Public API

// Gọi trong onlineStart() — đăng ký HTTP routes (hoạt động cả OFFLINE lẫn ONLINE)
void otaSetupRoutes() {
    server.on("/ota", HTTP_GET, handleOTAPage);
    server.on("/ota/info", HTTP_GET, handleOTAInfo);
    server.on("/ota/upload", HTTP_POST, handleOTAUpload, handleOTAUploadData);
    Serial.println(F("[OTA] Web OTA route sẵn sàng tại /ota"));
}

// Gọi trong onlineLoop() — handle ArduinoOTA (mDNS)
void otaLoop() {
    // Khởi tạo ArduinoOTA lần đầu khi WiFi STA kết nối thành công
    if (!_arduinoOTASetup && WiFi.status() == WL_CONNECTED) {
        _setupArduinoOTA();
    }

    // Handle ArduinoOTA events
    if (_arduinoOTASetup) {
        ArduinoOTA.handle();
    }
}
